cmake_minimum_required(VERSION 3.16...3.28 FATAL_ERROR)

AddExe(gamma_correction glfw.icns gamma_correction.cpp)

target_link_libraries(gamma_correction PRIVATE stbimage assimp)

target_link_directories(gamma_correction PRIVATE /opt/homebrew/lib/)

target_include_directories(gamma_correction PRIVATE ${CMAKE_SOURCE_DIR}/deps ${CMAKE_SOURCE_DIR}/StbImage /opt/homebrew/include/)

add_custom_command(
    TARGET gamma_correction POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_directory
    ${CMAKE_CURRENT_SOURCE_DIR}/resources/
    ${CMAKE_CURRENT_BINARY_DIR}/gamma_correction.app/Contents/Resources
)
