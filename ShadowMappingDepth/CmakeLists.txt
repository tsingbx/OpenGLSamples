cmake_minimum_required(VERSION 3.16...3.28 FATAL_ERROR)

AddExe(shadow_mapping_depth glfw.icns shadow_mapping_depth.cpp)

target_link_libraries(shadow_mapping_depth PRIVATE stbimage assimp)

target_link_directories(shadow_mapping_depth PRIVATE /opt/homebrew/lib/)

target_include_directories(shadow_mapping_depth PRIVATE ${CMAKE_SOURCE_DIR}/deps ${CMAKE_SOURCE_DIR}/StbImage /opt/homebrew/include/)

add_custom_command(
    TARGET shadow_mapping_depth POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_directory
    ${CMAKE_CURRENT_SOURCE_DIR}/resources/
    ${CMAKE_CURRENT_BINARY_DIR}/shadow_mapping_depth.app/Contents/Resources
)
