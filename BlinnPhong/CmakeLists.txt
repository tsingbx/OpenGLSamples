cmake_minimum_required(VERSION 3.16...3.28 FATAL_ERROR)

AddExe(blinn_phong glfw.icns blinn_phong.cpp)

target_link_libraries(blinn_phong PRIVATE stbimage assimp)

target_link_directories(blinn_phong PRIVATE /opt/homebrew/lib/)

target_include_directories(blinn_phong PRIVATE ${CMAKE_SOURCE_DIR}/deps ${CMAKE_SOURCE_DIR}/StbImage /opt/homebrew/include/)

add_custom_command(
    TARGET blinn_phong POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_directory
    ${CMAKE_CURRENT_SOURCE_DIR}/resources/
    ${CMAKE_CURRENT_BINARY_DIR}/blinn_phong.app/Contents/Resources
)
